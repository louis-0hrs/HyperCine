cmake_minimum_required(VERSION 3.25)
project(HyperCineWindowsReview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REVIEW_SOURCE "" CACHE PATH "Unmodified exact-head PR checkout")
set(REVIEW_KIND "" CACHE STRING "hashing or capability")
if(NOT EXISTS "${REVIEW_SOURCE}/include/hypercine/filmscan3.h")
    message(FATAL_ERROR "REVIEW_SOURCE must name a HyperCine checkout")
endif()
set(HYPERCINE_BUILD_STANDALONE ON CACHE BOOL "" FORCE)
set(HYPERCINE_BUILD_UPSTREAM OFF CACHE BOOL "" FORCE)
set(HYPERCINE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(HYPERCINE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(HYPERCINE_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(HYPERCINE_ENABLE_METAL OFF CACHE BOOL "" FORCE)
set(HYPERCINE_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(HYPERCINE_REQUIRE_EXTERNAL_DECODE OFF CACHE BOOL "" FORCE)
add_subdirectory("${REVIEW_SOURCE}" reviewed EXCLUDE_FROM_ALL)
enable_testing()
if(REVIEW_KIND STREQUAL "hashing")
    add_executable(review_hashing review_hashing.cpp)
    target_link_libraries(review_hashing PRIVATE hypercine_standalone)
    target_include_directories(review_hashing PRIVATE "${REVIEW_SOURCE}/tests")
    add_test(NAME windows_hashing_independent COMMAND review_hashing
        "${REVIEW_SOURCE}/tests/fixtures/macos-filmscan3/flat.cfhd")
elseif(REVIEW_KIND STREQUAL "capability")
    add_library(review_capability INTERFACE)
    target_link_libraries(review_capability INTERFACE hypercine_standalone)
    target_include_directories(review_capability INTERFACE "${REVIEW_SOURCE}/src/core")
    add_executable(review_frame_rates review_frame_rates.cpp)
    target_link_libraries(review_frame_rates PRIVATE review_capability)
    add_test(NAME windows_capability_frame_rates COMMAND review_frame_rates)
    add_executable(review_capability_matrix "${REVIEW_SOURCE}/tests/test_encoder_capability.cpp")
    target_link_libraries(review_capability_matrix PRIVATE review_capability)
    add_test(NAME windows_capability_matrix COMMAND review_capability_matrix)
else()
    message(FATAL_ERROR "REVIEW_KIND must be hashing or capability")
endif()
